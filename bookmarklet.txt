function analyze(c){
 const closes=c.map(x=>x.close), n=c.length, last=c[n-1], r=rsi(closes), e9=ema(closes.slice(-60),9), e21=ema(closes.slice(-60),21);
 const look=8, prev=c.slice(n-1-look,n-1); const low=Math.min(...prev.map(x=>x.low)), high=Math.max(...prev.map(x=>x.high));
 const bullSweep=last.low<low && last.close>low, bearSweep=last.high>high && last.close<high;
 const a=c[n-3], b=c[n-2], z=c[n-1];
 const bullFvg=b.low>a.high, bearFvg=b.high<a.low;
 let up=0,down=0;
